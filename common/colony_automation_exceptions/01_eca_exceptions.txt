# Enhanced Colony Automation AI - Emergency Exceptions
# 1) Build pop growth buildings (gene clinic, robot assembly) even without free pops
# 2) Proactive district/zone building: target 1 open job at all times
# 3) Deficit-reactive building: build production even without free pops if empire has deficit
# 4) Auto-upgrade existing buildings on ECA planets
# 5) Build precinct houses and strongholds on low-stability planets

# ===== BUILDING UPGRADES =====
# Ensure buildings on ECA planets get upgraded when the tech is available.
# The 2x stockpile checks on new construction can block upgrades, so this
# separate upgrade block fires unconditionally for all ECA designations.
automate_eca_upgrade_buildings = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		exists = owner
		owner = { is_regular_empire = yes }
	}

	emergency = yes
	category = "planet_automation_upgrade_buildings"

	buildings = {
		# District capacity upgrades (Industrial Mining Site → Core Drilling Array → etc.)
		mining_districts = {
			building = building_mining_districts_1
		}
		generator_districts = {
			building = building_generator_districts_1
		}
		farming_districts = {
			building = building_farming_districts_1
		}
		# Research labs
		research_lab = {
			building = building_research_lab_1
		}
		physics_lab = {
			building = building_physics_lab_1
		}
		biolab = {
			building = building_biolab_1
		}
		engineering = {
			building = building_engineering_facility_1
		}
		# Production
		foundry = {
			building = building_foundry_1
		}
		factory = {
			building = building_factory_1
		}
		# Amenity / support
		holo = {
			building = building_holo_theatres
		}
		gene_clinic = {
			building = building_medical_1
		}
		# Unity
		admin = {
			building = building_bureaucratic_1
		}
		temple = {
			building = building_temple
		}
		# Strategic
		refinery = {
			building = building_refinery
		}
		chemical = {
			building = building_chemical_plant
		}
		crystal = {
			building = building_crystal_plant
		}
		# Stability
		precinct = {
			building = building_precinct_house
		}
		stronghold = {
			building = building_stronghold
		}
	}
}

# ===== POP GROWTH: Robot Assembly =====
# Build robot assembly plant on ECA planets if empire can afford 2x the alloy upkeep.
# Roboticist costs 2 alloys/month, so require 4+ alloy surplus.
automate_eca_robot_assembly = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
		num_pops > 0
		can_assemble_robot = yes
		# Don't build pop growth if pops are already unemployed
		NOT = { any_owned_pop_group = { is_unemployed = yes } }
		exists = owner
		owner = {
			is_regular_empire = yes
			NOT = { has_policy_flag = robots_outlawed }
			NOT = { has_ascension_perk = ap_engineered_evolution }
			eca_can_afford_robot_assembly_upkeep = yes
			eca_can_afford_build_cost_400 = yes
		}
	}

	emergency = yes
	category = "planet_automation_pop_assembly"

	buildings = {
		assembly = {
			building = building_robot_assembly_plant
		}
	}
}

# ===== LOW STABILITY: Precinct House =====
# Build a precinct house when planet stability drops below 50 (penalty threshold).
# Enforcers reduce crime and boost stability. Caps at 1 unless stability is very low.
automate_eca_stability_precinct = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
		num_pops > 0
		planet_stability < 50
		NOT = { has_forbidden_jobs = enforcer }
		exists = owner
		owner = {
			is_regular_empire = yes
			is_gestalt = no
			eca_can_afford_energy_upkeep = yes
			eca_can_afford_build_cost_400 = yes
		}
		# Cap at 1 precinct unless stability is critically low
		OR = {
			num_buildings = { type = building_precinct_house value < 1 }
			AND = {
				planet_stability < 25
				num_buildings = { type = building_precinct_house value < 2 }
			}
		}
	}

	emergency = yes
	category = "planet_automation_crime"

	buildings = {
		precinct = {
			building = building_precinct_house
		}
	}
}

# ===== PROACTIVE DISTRICTS =====
# Build the next district/zone when:
#   A) All pops have jobs and no relevant jobs are open (proactive), OR
#   B) Empire has a deficit in the relevant resource (deficit-reactive)
# This keeps ~1 job opening ready for the next pop, and also responds to deficits.

automate_eca_proactive_mining = {
	available = {
		has_designation = col_eca_mining
		free_district_slots > 0
		num_pops > 0
		exists = owner
		owner = { eca_can_afford_build_cost_300 = yes }
		OR = {
			# No open mining jobs — build whether pops are unemployed or all employed
			NOT = { has_available_jobs = miner }
			# Deficit-reactive: empire needs minerals
			owner = { eca_has_mineral_deficit = yes }
		}
	}

	emergency = yes
	category = "planet_automation_designation_construction"

	prio_districts = {
		district_mining
		district_mining_uncapped
	}
}

# ===== DEFICIT-REACTIVE BUILDING =====
# Build production buildings even without free pops when the empire has a deficit.
# The building sits idle until a pop grows into the job, but for deficits the
# tradeoff is worth it — the pop immediately starts producing what's needed.
# These fire on ANY ECA designation (a mining world should build a factory if
# the empire desperately needs consumer goods).

automate_eca_deficit_alloys = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
		num_pops > 0
		exists = owner
		owner = {
			is_regular_empire = yes
			eca_has_alloy_deficit = yes
			eca_can_afford_foundry_upkeep = yes
			eca_can_afford_build_cost_400 = yes
		}
		# Prefer dedicated foundry/industrial worlds — only build off-designation if none have room
		OR = {
			has_designation = col_eca_foundry
			has_designation = col_eca_industrial
			has_designation = col_foundry
			has_designation = col_industrial
			owner = {
				NOT = {
					any_owned_planet = {
						OR = {
							has_designation = col_eca_foundry
							has_designation = col_eca_industrial
							has_designation = col_foundry
							has_designation = col_industrial
						}
						free_building_slots > 0
					}
				}
			}
		}
	}

	emergency = yes
	category = "planet_automation_designation_construction"

	buildings = {
		alloy_foundry = {
			building = building_foundry_1
		}
	}
}
