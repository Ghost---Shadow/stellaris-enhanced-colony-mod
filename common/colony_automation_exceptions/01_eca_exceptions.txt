# Enhanced Colony Automation AI - Emergency Exceptions
# 1) Build pop growth buildings (gene clinic) even without free pops
# 2) Proactive district/zone building: target 1 open job at all times

# ===== POP GROWTH: Gene Clinic =====
# Build gene clinic as emergency on ECA planets regardless of free pops
automate_eca_gene_clinic = {
	available = {
		exists = owner
		owner = {
			is_regular_empire = yes
			is_synthetic_empire = no
		}
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
	}

	emergency = yes
	category = "planet_automation_pop_assembly"

	buildings = {
		gene_clinic = {
			building = building_medical_1
		}
	}
}

# ===== PROACTIVE DISTRICTS =====
# Build the next district/zone when all pops have jobs and no relevant jobs are open.
# This keeps ~1 job opening ready for the next pop that grows.

automate_eca_proactive_mining = {
	available = {
		has_designation = col_eca_mining
		NOT = { any_owned_pop_group = { is_unemployed = yes } }
		NOT = { has_available_jobs = miner }
		free_district_slots > 0
	}

	emergency = yes
	category = "planet_automation_designation_construction"

	prio_districts = {
		district_mining
		district_mining_uncapped
	}
}
