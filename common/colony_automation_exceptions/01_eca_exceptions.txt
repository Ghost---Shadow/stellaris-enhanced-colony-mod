# Enhanced Colony Automation AI - Emergency Exceptions
# 1) Build pop growth buildings (gene clinic, robot assembly) even without free pops
# 2) Proactive district/zone building: target 1 open job at all times
# 3) Deficit-reactive building: build production even without free pops if empire has deficit

# ===== POP GROWTH: Robot Assembly =====
# Build robot assembly plant on ECA planets if empire can afford 2x the alloy upkeep.
# Roboticist costs 2 alloys/month, so require 4+ alloy surplus.
automate_eca_robot_assembly = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
		can_assemble_robot = yes
		exists = owner
		owner = {
			is_regular_empire = yes
			NOT = { has_policy_flag = robots_outlawed }
			NOT = { has_ascension_perk = ap_engineered_evolution }
			eca_can_afford_robot_assembly_upkeep = yes
		}
	}

	emergency = yes
	category = "planet_automation_pop_assembly"

	buildings = {
		assembly = {
			building = building_robot_assembly_plant
		}
	}
}

# ===== PROACTIVE DISTRICTS =====
# Build the next district/zone when:
#   A) All pops have jobs and no relevant jobs are open (proactive), OR
#   B) Empire has a deficit in the relevant resource (deficit-reactive)
# This keeps ~1 job opening ready for the next pop, and also responds to deficits.

automate_eca_proactive_mining = {
	available = {
		has_designation = col_eca_mining
		free_district_slots > 0
		OR = {
			# Proactive: all employed, no open mining jobs
			AND = {
				NOT = { any_owned_pop_group = { is_unemployed = yes } }
				NOT = { has_available_jobs = miner }
			}
			# Deficit-reactive: empire needs minerals
			owner = { eca_has_mineral_deficit = yes }
		}
	}

	emergency = yes
	category = "planet_automation_designation_construction"

	prio_districts = {
		district_mining
		district_mining_uncapped
	}
}

# ===== DEFICIT-REACTIVE BUILDING =====
# Build production buildings even without free pops when the empire has a deficit.
# The building sits idle until a pop grows into the job, but for deficits the
# tradeoff is worth it — the pop immediately starts producing what's needed.
# These fire on ANY ECA designation (a mining world should build a factory if
# the empire desperately needs consumer goods).

automate_eca_deficit_alloys = {
	available = {
		OR = {
			has_designation = col_eca_capital
			has_designation = col_eca_mining
			has_designation = col_eca_generator
			has_designation = col_eca_farming
			has_designation = col_eca_foundry
			has_designation = col_eca_factory
			has_designation = col_eca_research
			has_designation = col_eca_industrial
		}
		free_building_slots > 0
		exists = owner
		owner = {
			is_regular_empire = yes
			eca_has_alloy_deficit = yes
			eca_can_afford_foundry_upkeep = yes
		}
	}

	emergency = yes
	category = "planet_automation_designation_construction"

	buildings = {
		alloy_foundry = {
			building = building_foundry_1
		}
	}
}
