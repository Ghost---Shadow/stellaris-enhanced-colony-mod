# ===== ENHANCED CAPITAL =====

automate_eca_capital = {
	category = "planet_automation_designation_construction"

	available = {
		has_designation = col_eca_capital
	}

	prio_districts = {
		district_city
		district_arcology_housing
		district_rw_city
		district_hab_housing
	}

	prio_zones = {
		zone_industrial
		zone_research_unity
		zone_foundry
		zone_energy
		zone_food
		zone_minerals
	}

	buildings = {
		1 = {
			building = building_embassy
		}

		2 = {
			building = building_autochthon_monument
		}

		2_corp = {
			building = building_corporate_monument
		}

		3 = {
			building = building_vultaum_reality_computer
		}

		4 = {
			building = building_institute
		}

		5 = {
			building = building_archaeostudies_faculty
		}

		6 = {
			building = building_autocurating_vault
		}

		6_spir = {
			building = building_citadel_of_faith
		}

		6_corp = {
			building = building_corporate_vault
		}

		# Gene clinic for pop growth + amenities
		7 = {
			building = building_medical_1
			available = {
				owner = { is_synthetic_empire = no }
			}
		}

		# Alloy foundry - capitals should produce alloys early
		8 = {
			building = building_foundry_1
			available = {
				exists = owner
				owner = {
					OR = {
						eca_has_alloy_deficit = yes
						eca_is_early_game = yes
					}
				}
			}
		}

		# Consumer goods factory
		9 = {
			building = building_factory_1
			available = {
				exists = owner
				owner = {
					country_uses_consumer_goods = yes
					OR = {
						eca_has_consumer_goods_deficit = yes
						eca_is_early_game = yes
					}
				}
			}
		}

		# Research lab
		10 = {
			building = building_research_lab_1
		}

		# Holo theatre for amenity coverage
		11 = {
			building = building_holo_theatres
			available = {
				free_amenities < 5
			}
		}

		# Strategic resource buildings (deficit-conditional)
		12 = {
			building = building_refinery
			available = {
				exists = owner
				owner = { eca_has_exotic_gas_deficit = yes }
			}
		}

		13 = {
			building = building_chemical_plant
			available = {
				exists = owner
				owner = { eca_has_volatile_mote_deficit = yes }
			}
		}

		14 = {
			building = building_crystal_plant
			available = {
				exists = owner
				owner = { eca_has_rare_crystal_deficit = yes }
			}
		}
	}
}
